# Phase 0-1 MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first working MVP of `llm-inference-delivery-lab`: a documented project baseline plus a Docker Compose based 14B vLLM serving stack with FastAPI gateway, streaming client, benchmark harness, and basic metrics.

**Architecture:** This plan covers only Phase 0 and Phase 1 from the design spec. The repository will expose vLLM as the upstream inference engine, wrap it with a focused FastAPI gateway, provide a small async benchmark harness, and document the experiment workflow. Later phases for quantization, observability stack, K8s, free GPU evaluation, heterogeneous toolchains, and incident playbooks must be separate plans.

**Tech Stack:** Python 3.11+, FastAPI, Uvicorn, httpx, pydantic-settings, prometheus-client, pytest, pytest-asyncio, Docker Compose, vLLM OpenAI-compatible API.

The full design spec contains multiple independent subsystems. This plan intentionally implements only:

- Phase 0: repository baseline, documentation templates, scenario/SLO matrix.
- Phase 1: single-machine inference delivery MVP using vLLM, FastAPI gateway, Docker Compose, streaming client, benchmark harness, and basic metrics.

Separate future plans are required for:

- Phase 2: quantization and parameter tuning.
- Phase 3: Prometheus/Grafana/Loki observability stack and reliability tests.
- Phase 4: Kubernetes/Helm deployment.
- Phase 5: free GPU platform evaluation.
- Phase 6: heterogeneous accelerator migration design.

```python
from collections.abc import AsyncIterator
from typing import Any

import httpx
from fastapi import HTTPException
from starlette.responses import StreamingResponse


def build_chat_payload(body: dict[str, Any], default_model: str) -> dict[str, Any]:
    messages = body.get("messages")
    if not messages:
        raise ValueError("messages must not be empty")

    payload = dict(body)
    payload["model"] = payload.get("model") or default_model
    payload["messages"] = messages
    payload["stream"] = bool(payload.get("stream", False))
    return payload


async def post_chat_completion(
    *,
    base_url: str,
    payload: dict[str, Any],
    timeout_seconds: int,
) -> dict[str, Any]:
    try:
        async with httpx.AsyncClient(timeout=timeout_seconds) as client:
            response = await client.post(f"{base_url}/v1/chat/completions", json=payload)
            response.raise_for_status()
            return response.json()
    except httpx.HTTPStatusError as exc:
        raise HTTPException(status_code=exc.response.status_code, detail=exc.response.text) from exc
    except httpx.TimeoutException as exc:
        raise HTTPException(status_code=504, detail="upstream request timed out") from exc
    except httpx.HTTPError as exc:
        raise HTTPException(status_code=502, detail="upstream request failed") from exc


async def stream_chat_completion(
    *,
    base_url: str,
    payload: dict[str, Any],
    timeout_seconds: int,
) -> StreamingResponse:
    async def event_stream() -> AsyncIterator[bytes]:
        try:
            async with httpx.AsyncClient(timeout=timeout_seconds) as client:
                async with client.stream(
                    "POST",
                    f"{base_url}/v1/chat/completions",
                    json=payload,
                ) as response:
                    response.raise_for_status()
                    async for chunk in response.aiter_bytes():
                        yield chunk
        except httpx.HTTPError as exc:
            yield f'data: {{"error":"{str(exc)}"}}\n\n'.encode()

    return StreamingResponse(event_stream(), media_type="text/event-stream")
```

Replace `gateway/app/main.py` with:

```python
import time
from typing import Any

from fastapi import Depends, FastAPI, Header, HTTPException

from gateway.app.config import Settings, get_settings
from gateway.app.metrics import REQUEST_LATENCY_SECONDS, REQUESTS_TOTAL, metrics_response
from gateway.app.upstream import build_chat_payload, post_chat_completion, stream_chat_completion

app = FastAPI(title="LLM Inference Gateway")


@app.get("/health")
def health() -> dict[str, str]:
    REQUESTS_TOTAL.labels(route="/health", status="ok").inc()
    return {"status": "ok"}


@app.get("/metrics")
def metrics():
    return metrics_response()


def require_api_key(
    authorization: str | None = Header(default=None),
    settings: Settings = Depends(get_settings),
) -> None:
    expected = f"Bearer {settings.gateway_api_key}"
    if authorization != expected:
        raise HTTPException(status_code=401, detail="missing or invalid API key")


@app.post("/v1/chat/completions")
async def chat_completions(
    body: dict[str, Any],
    _: None = Depends(require_api_key),
    settings: Settings = Depends(get_settings),
):
    started = time.perf_counter()
    route = "/v1/chat/completions"
    try:
        payload = build_chat_payload(body, default_model=settings.default_model)
        if payload["stream"]:
            REQUESTS_TOTAL.labels(route=route, status="stream").inc()
            return await stream_chat_completion(
                base_url=settings.vllm_base_url,
                payload=payload,
                timeout_seconds=settings.stream_timeout_seconds,
            )
        result = await post_chat_completion(
            base_url=settings.vllm_base_url,
            payload=payload,
            timeout_seconds=settings.request_timeout_seconds,
        )
        REQUESTS_TOTAL.labels(route=route, status="ok").inc()
        return result
    except ValueError as exc:
        REQUESTS_TOTAL.labels(route=route, status="bad_request").inc()
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    finally:
        REQUEST_LATENCY_SECONDS.labels(route=route).observe(time.perf_counter() - started)
```

```python
import json
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class WorkloadItem:
    id: str
    prompt: str
    max_tokens: int = 512


def load_workload(path: Path) -> list[WorkloadItem]:
    items: list[WorkloadItem] = []
    with path.open("r", encoding="utf-8") as handle:
        for line_number, line in enumerate(handle, start=1):
            if not line.strip():
                continue
            raw = json.loads(line)
            prompt = raw.get("prompt")
            if not prompt:
                raise ValueError(f"line {line_number}: prompt is required")
            items.append(
                WorkloadItem(
                    id=str(raw.get("id", line_number)),
                    prompt=prompt,
                    max_tokens=int(raw.get("max_tokens", 512)),
                )
            )
    if not items:
        raise ValueError("workload must contain at least one item")
    return items
```

```python
import math


def percentile(values: list[float], percent: int) -> float:
    if not values:
        return 0.0
    ordered = sorted(values)
    rank = math.ceil((percent / 100) * len(ordered)) - 1
    return ordered[max(0, min(rank, len(ordered) - 1))]


def summarize_latencies(values: list[float]) -> dict[str, float | int]:
    return {
        "count": len(values),
        "p50": percentile(values, 50),
        "p95": percentile(values, 95),
        "p99": percentile(values, 99),
        "min": min(values) if values else 0.0,
        "max": max(values) if values else 0.0,
        "avg": sum(values) / len(values) if values else 0.0,
    }
```

```python
import asyncio
import json
import time
from dataclasses import asdict, dataclass
from pathlib import Path

import httpx

from benchmark.harness.metrics import summarize_latencies
from benchmark.harness.workload import WorkloadItem


@dataclass(frozen=True)
class RequestResult:
    id: str
    ok: bool
    latency_seconds: float
    ttft_seconds: float | None
    output_bytes: int
    error: str | None = None


async def run_one(
    *,
    client: httpx.AsyncClient,
    base_url: str,
    api_key: str,
    item: WorkloadItem,
    stream: bool,
) -> RequestResult:
    payload = {
        "messages": [{"role": "user", "content": item.prompt}],
        "max_tokens": item.max_tokens,
        "stream": stream,
    }
    started = time.perf_counter()
    first_chunk_at: float | None = None
    output_bytes = 0
    try:
        if stream:
            async with client.stream(
                "POST",
                f"{base_url}/v1/chat/completions",
                headers={"Authorization": f"Bearer {api_key}"},
                json=payload,
            ) as response:
                response.raise_for_status()
                async for chunk in response.aiter_bytes():
                    if first_chunk_at is None:
                        first_chunk_at = time.perf_counter()
                    output_bytes += len(chunk)
        else:
            response = await client.post(
                f"{base_url}/v1/chat/completions",
                headers={"Authorization": f"Bearer {api_key}"},
                json=payload,
            )
            response.raise_for_status()
            output_bytes = len(response.content)
        ended = time.perf_counter()
        return RequestResult(
            id=item.id,
            ok=True,
            latency_seconds=ended - started,
            ttft_seconds=(first_chunk_at - started) if first_chunk_at else None,
            output_bytes=output_bytes,
        )
    except Exception as exc:
        ended = time.perf_counter()
        return RequestResult(
            id=item.id,
            ok=False,
            latency_seconds=ended - started,
            ttft_seconds=None,
            output_bytes=output_bytes,
            error=str(exc),
        )


async def run_benchmark(
    *,
    base_url: str,
    api_key: str,
    workload: list[WorkloadItem],
    concurrency: int,
    request_count: int,
    stream: bool,
    timeout_seconds: int,
) -> dict:
    selected = [workload[index % len(workload)] for index in range(request_count)]
    semaphore = asyncio.Semaphore(concurrency)

    async with httpx.AsyncClient(timeout=timeout_seconds) as client:
        async def guarded(item: WorkloadItem) -> RequestResult:
            async with semaphore:
                return await run_one(
                    client=client,
                    base_url=base_url,
                    api_key=api_key,
                    item=item,
                    stream=stream,
                )

        started = time.perf_counter()
        results = await asyncio.gather(*(guarded(item) for item in selected))
        total_seconds = time.perf_counter() - started

    successful = [result for result in results if result.ok]
    latencies = [result.latency_seconds for result in successful]
    ttfts = [result.ttft_seconds for result in successful if result.ttft_seconds is not None]
    return {
        "summary": {
            "total_requests": len(results),
            "successful_requests": len(successful),
            "failed_requests": len(results) - len(successful),
            "total_seconds": total_seconds,
            "requests_per_second": len(results) / total_seconds if total_seconds > 0 else 0.0,
            "latency_seconds": summarize_latencies(latencies),
            "ttft_seconds": summarize_latencies(ttfts),
        },
        "results": [asdict(result) for result in results],
    }


def write_result(path: Path, result: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(result, ensure_ascii=False, indent=2), encoding="utf-8")
```
